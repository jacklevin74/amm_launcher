% Key Mechanisms
\section{Key Mechanisms Verified}

\subsection{USDC Operations (Deposit/Withdraw)}
\begin{itemize}
    \item Maintain $\text{total\_usdc} = \text{virtual\_usdc} + (\text{real\_usdc} \times 1000)$ constant
    \item E6 to E9 normalization working correctly
    \item Virtual reserve adjusts inversely to real USDC transfers
\end{itemize}

\subsection{XNT Operations (Deposit/Withdraw)}
\begin{itemize}
    \item Both XNT and USDC reserves adjust proportionally
    \item Ratio of increase/decrease identical for both reserves
    \item Price $= \frac{\text{usdc\_reserve}}{\text{xnt\_reserve}}$ remains constant
\end{itemize}

\subsection{Ceiling Reserve Funding}
\begin{itemize}
    \item Completely separate from AMM pool reserves
    \item Zero impact on pool pricing or constant $k$
    \item Used exclusively for \$2.00 price ceiling defense
\end{itemize}

% Technical Details
\section{Technical Implementation}

\subsection{Constant Product Formula}
The AMM uses the constant product formula:
\begin{equation}
k = x \times y
\end{equation}
where:
\begin{itemize}
    \item $x$ = XNT reserve
    \item $y$ = USDC reserve
    \item $k$ = constant (updates when reserves change via price-neutral operations)
\end{itemize}

\subsection{Price Calculation}
\begin{equation}
\text{Price} = \frac{\text{USDC Reserve}}{\text{XNT Reserve}}
\end{equation}

For USDC operations:
\begin{equation}
\text{Total USDC} = \text{Virtual USDC (e9)} + (\text{Real USDC (e6)} \times 1000)
\end{equation}

\subsection{E6 to E9 Normalization}
USDC uses 6 decimals (e6) while the internal system uses 9 decimals (e9). The conversion is:
\begin{equation}
\text{USDC}_{e9} = \text{USDC}_{e6} \times 1000
\end{equation}
